def parse_summary(path):
    seizure_info = {}
    with open(path) as f:
        content = f.read()
    blocks = content.split('File Name:')[1:]
    for block in blocks:
        lines = block.strip().split('\n')
        fname = lines[0].strip()
        seizure_info[fname] = []
        start = end = 0
        for line in lines:
            m = re.search(r'Number of Seizures in File:\s*(\d+)', line)
            if m:
                pass  # just for bookkeeping
            ms = re.search(r'Seizure(?:\s+\d+)?\s+Start\s+Time:\s*(\d+)', line)
            me = re.search(r'Seizure(?:\s+\d+)?\s+End\s+Time:\s*(\d+)', line)
            if ms:
                start = int(ms.group(1))
            if me:
                end = int(me.group(1))
                seizure_info[fname].append((start, end))
    return seizure_info

def load_edf(path, n_channels=18):
    f = pyedflib.EdfReader(path)
    ch_names = f.getSignalLabels()
    fs       = int(f.getSampleFrequency(0))
    n_samp   = f.getNSamples()[0]
    n_ch     = min(f.signals_in_file, n_channels)
    sigs = np.zeros((n_ch, n_samp))
    for i in range(n_ch):
        sigs[i] = f.readSignal(i)
    f.close()
    return sigs, ch_names[:n_ch], fs

FREQ_BANDS = {
    'Delta (0.5-4)' : (0.5,  4),
    'Theta (4-8)'   : (4,    8),
    'Alpha (8-13)'  : (8,   13),
    'Beta (13-30)'  : (13,  30),
    'Gamma (30-100)': (30, 100),
}

def band_power(sig, fs, flo, fhi):
    freqs, psd = welch(sig, fs=fs, nperseg=min(256, len(sig)))
    idx = np.logical_and(freqs >= flo, freqs <= fhi)
    return np.trapz(psd[idx], freqs[idx])

# CELL 6 — Preprocessing Pipeline
# ============================================================
class EEGPreprocessor:
    def __init__(self, fs=256, lo=0.5, hi=100, notch=60):
        self.fs = fs; self.lo = lo; self.hi = hi; self.notch = notch

    def bandpass(self, data, order=4):
        lo_n = self.lo / (self.fs / 2)
        hi_n = min(self.hi / (self.fs / 2), 0.999)
        b, a = butter(order, [lo_n, hi_n], btype='band')
        return filtfilt(b, a, data)

    def notch_filter(self, data, order=4):
        n = self.notch / (self.fs / 2)
        b, a = butter(order, [n - 0.02, n + 0.02], btype='bandstop')
        return filtfilt(b, a, data)

    def normalize(self, data):
        mu = data.mean(axis=1, keepdims=True)
        sd = data.std(axis=1, keepdims=True) + 1e-8
        return (data - mu) / sd

    def preprocess(self, data):
        out = np.zeros_like(data)
        for i in range(data.shape[0]):
            ch = self.bandpass(data[i])
            ch = self.notch_filter(ch)
            out[i] = ch
        return self.normalize(out)

class SpikeEncoder(nn.Module):
    """Convert analog EEG to binary spike train via threshold."""

    def __init__(self, threshold=0.5):
        super().__init__()
        self.threshold = nn.Parameter(torch.tensor(threshold))

    def forward(self, x):
        # x: (B, T, C)
        spikes = (x.abs() > self.threshold).float()
        mem = torch.zeros_like(x)
        for t in range(x.shape[1]):
            prev = mem[:, t - 1] if t > 0 else torch.zeros_like(x[:, 0])
            mem[:, t] = 0.9 * prev + x[:, t]
        return spikes + 0.1 * torch.tanh(mem)


class SpikingConvBlock(nn.Module):
    def __init__(self, in_ch, out_ch, k=5):
        super().__init__()
        self.conv = nn.Conv1d(in_ch, out_ch, k, padding=k // 2)
        self.bn   = nn.BatchNorm1d(out_ch)
        self.pool = nn.MaxPool1d(2)

    def forward(self, x):
        return F.relu(self.pool(self.bn(self.conv(x))))


class SCT(nn.Module):
    def __init__(self, n_ch=18, T=1024, n_classes=1,
                 d_model=64, n_heads=4, n_layers=2, dropout=0.2):
        super().__init__()
        self.spike_enc = SpikeEncoder()
        self.conv1 = SpikingConvBlock(n_ch,    32, k=7)
        self.conv2 = SpikingConvBlock(32,      64, k=5)
        self.conv3 = SpikingConvBlock(64, d_model, k=3)

        enc_layer = nn.TransformerEncoderLayer(
            d_model=d_model, nhead=n_heads,
            dim_feedforward=d_model * 4, dropout=dropout, batch_first=True
        )
        self.transformer = nn.TransformerEncoder(enc_layer, num_layers=n_layers)
        self.cls_token   = nn.Parameter(torch.zeros(1, 1, d_model))
        self.pool_fc     = nn.Sequential(
            nn.Linear(d_model, 32), nn.ReLU(), nn.Dropout(dropout),
            nn.Linear(32, n_classes), nn.Sigmoid()
        )

    def forward(self, x):
        # x: (B, T, C)
        x   = self.spike_enc(x)
        x   = x.permute(0, 2, 1)     # (B, C, T)
        x   = self.conv1(x)
        x   = self.conv2(x)
        x   = self.conv3(x)
        x   = x.permute(0, 2, 1)     # (B, T', d_model)
        cls = self.cls_token.expand(x.size(0), -1, -1)
        x   = torch.cat([cls, x], dim=1)
        x   = self.transformer(x)
        out = self.pool_fc(x[:, 0])
        return out.squeeze(-1)

def train_model(model, train_loader, val_loader, epochs=30, lr=1e-3, name='Model'):
    optimizer = torch.optim.Adam(model.parameters(), lr=lr, weight_decay=1e-4)
    scheduler = torch.optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=epochs)

    n_pos = sum(y.sum().item() for _, y in train_loader)
    n_neg = sum((1 - y).sum().item() for _, y in train_loader)
    pos_w = torch.tensor(n_neg / max(n_pos, 1)).to(device)
    criterion = nn.BCELoss(reduction='none')

    history = {'train_loss': [], 'val_loss': [], 'val_auc': []}
    best_auc, best_state = 0, None

    print(f"  Training {name} for {epochs} epochs…")
    for ep in range(1, epochs + 1):
        model.train(); ep_loss = 0
        for Xb, yb in train_loader:
            Xb, yb = Xb.to(device), yb.to(device)
            optimizer.zero_grad()
            pred = model(Xb)
            w    = torch.where(yb == 1, pos_w, torch.ones_like(pos_w))
            loss = (criterion(pred, yb) * w).mean()
            loss.backward(); optimizer.step()
            ep_loss += loss.item()
        scheduler.step()

        model.eval(); vloss = 0; vp, vy = [], []
        with torch.no_grad():
            for Xb, yb in val_loader:
                Xb, yb = Xb.to(device), yb.to(device)
                pred   = model(Xb)
                w      = torch.where(yb == 1, pos_w, torch.ones_like(pos_w))
                vloss += (criterion(pred, yb) * w).mean().item()
                vp.extend(pred.cpu().numpy()); vy.extend(yb.cpu().numpy())
        try:    vauc = roc_auc_score(vy, vp)
        except: vauc = 0.5

        history['train_loss'].append(ep_loss / len(train_loader))
        history['val_loss'].append(vloss / len(val_loader))
        history['val_auc'].append(vauc)

        if vauc > best_auc:
            best_auc = vauc; best_state = copy.deepcopy(model.state_dict())
        if ep % 5 == 0:
            print(f"    Ep {ep:03d} | TrainLoss={ep_loss/len(train_loader):.4f} "
                  f"| ValLoss={vloss/len(val_loader):.4f} | ValAUC={vauc:.4f}")

    model.load_state_dict(best_state)
    print(f"  ✅  Best ValAUC = {best_auc:.4f}")
    return history

class GradientReversalFunction(Function):
    @staticmethod
    def forward(ctx, x, alpha):
        ctx.save_for_backward(alpha)
        return x.clone()

    @staticmethod
    def backward(ctx, grad):
        alpha, = ctx.saved_tensors
        return -alpha * grad, None


class GRL(nn.Module):
    def __init__(self):
        super().__init__()

    def forward(self, x, alpha=None):
        if alpha is None:
            alpha = torch.tensor(1.0)
        return GradientReversalFunction.apply(x, alpha.to(x.device))


class GRLSeizureModel(nn.Module):
    """EEG seizure detector with domain-adversarial training via GRL."""

    def __init__(self, n_ch=18, T=1024, d=64, dropout=0.3):
        super().__init__()
        self.feat = nn.Sequential(
            nn.Conv1d(n_ch, 32, 7, padding=3), nn.BatchNorm1d(32), nn.ReLU(), nn.MaxPool1d(2),
            nn.Conv1d(32,   64, 5, padding=2), nn.BatchNorm1d(64), nn.ReLU(), nn.MaxPool1d(2),
            nn.Conv1d(64,  128, 3, padding=1), nn.BatchNorm1d(128), nn.ReLU(), nn.AdaptiveAvgPool1d(16),
        )
        flat_dim = 128 * 16
        self.label_clf = nn.Sequential(
            nn.Linear(flat_dim, 256), nn.ReLU(), nn.Dropout(dropout),
            nn.Linear(256, 64),       nn.ReLU(), nn.Dropout(dropout),
            nn.Linear(64, 1),         nn.Sigmoid()
        )
        self.grl = GRL()
        self.domain_clf = nn.Sequential(
            nn.Linear(flat_dim, 128), nn.ReLU(), nn.Dropout(dropout),
            nn.Linear(128, 32),       nn.ReLU(),
            nn.Linear(32, 2)
        )

    def forward(self, x, alpha=1.0):
        # x: (B, T, C)
        x = x.permute(0, 2, 1)   # → (B, C, T)
        f = self.feat(x).view(x.size(0), -1)
        label_out  = self.label_clf(f)
        alpha_t    = torch.tensor(alpha, dtype=torch.float32)
        domain_in  = self.grl(f, alpha_t)
        domain_out = self.domain_clf(domain_in)
        return label_out.squeeze(-1), domain_out

def train_grl(model, train_loader, val_loader, epochs=30, lr=1e-3):
    optimizer  = torch.optim.Adam(model.parameters(), lr=lr, weight_decay=1e-4)
    scheduler  = torch.optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=epochs)
    lbl_crit   = nn.BCELoss()
    dom_crit   = nn.CrossEntropyLoss()

    history = {'train_loss': [], 'val_loss': [], 'val_auc': []}
    best_auc, best_state = 0, None

    print(f"  Training GRL for {epochs} epochs…")
    for ep in range(1, epochs + 1):
        model.train(); ep_loss = 0
        p     = ep / epochs
        alpha = 2.0 / (1 + np.exp(-10 * p)) - 1.0

        for Xb, yb in train_loader:
            Xb, yb = Xb.to(device), yb.to(device)
            dom_labels = torch.randint(0, 2, (Xb.size(0),)).to(device)
            optimizer.zero_grad()
            lbl_out, dom_out = model(Xb, alpha)
            loss = lbl_crit(lbl_out, yb) + 0.1 * dom_crit(dom_out, dom_labels)
            loss.backward(); optimizer.step()
            ep_loss += loss.item()
        scheduler.step()

        model.eval(); vloss = 0; vp, vy = [], []
        with torch.no_grad():
            for Xb, yb in val_loader:
                Xb, yb = Xb.to(device), yb.to(device)
                dom_labels = torch.randint(0, 2, (Xb.size(0),)).to(device)
                lbl_out, dom_out = model(Xb, alpha)
                vloss += (lbl_crit(lbl_out, yb) + 0.1 * dom_crit(dom_out, dom_labels)).item()
                vp.extend(lbl_out.cpu().numpy()); vy.extend(yb.cpu().numpy())
        try:    vauc = roc_auc_score(vy, vp)
        except: vauc = 0.5

        history['train_loss'].append(ep_loss / len(train_loader))
        history['val_loss'].append(vloss / len(val_loader))
        history['val_auc'].append(vauc)
        if vauc > best_auc:
            best_auc = vauc; best_state = copy.deepcopy(model.state_dict())
        if ep % 5 == 0:
            print(f"    Ep {ep:03d} | Loss={ep_loss/len(train_loader):.4f} | ValAUC={vauc:.4f} | α={alpha:.3f}")

    model.load_state_dict(best_state)
    print(f"  ✅  Best ValAUC = {best_auc:.4f}")
    return history

class FederatedModel(nn.Module):
    """Compact model for federated training across virtual clients."""

    def __init__(self, n_ch=18, T=1024, dropout=0.3):
        super().__init__()
        self.features = nn.Sequential(
            nn.Conv1d(n_ch, 32, 7, padding=3), nn.BatchNorm1d(32), nn.ReLU(), nn.MaxPool1d(4),
            nn.Conv1d(32,   64, 5, padding=2), nn.BatchNorm1d(64), nn.ReLU(), nn.MaxPool1d(4),
            nn.Conv1d(64,  128, 3, padding=1), nn.BatchNorm1d(128), nn.ReLU(), nn.AdaptiveAvgPool1d(8),
        )
        self.classifier = nn.Sequential(
            nn.Flatten(),
            nn.Linear(128 * 8, 128), nn.ReLU(), nn.Dropout(dropout),
            nn.Linear(128, 1), nn.Sigmoid()
        )

    def forward(self, x):
        x = x.permute(0, 2, 1)   # (B,T,C)→(B,C,T)
        return self.classifier(self.features(x)).squeeze(-1)


def add_dp_noise(model, sigma=0.01, clip_norm=1.0):
    """Differential privacy: clip gradients + add Gaussian noise."""
    with torch.no_grad():
        total_norm = sum(
            p.grad.data.norm(2).item() ** 2
            for p in model.parameters() if p.grad is not None
        ) ** 0.5
        clip_coef = clip_norm / max(total_norm, clip_norm)
        for p in model.parameters():
            if p.grad is not None:
                p.grad.data.mul_(clip_coef)
                p.grad.data.add_(torch.randn_like(p.grad.data) * sigma)


def federated_average(global_model, client_models):
    """FedAvg: average parameters from all clients."""
    global_dict = global_model.state_dict()
    for k in global_dict:
        stacked = torch.stack([cm.state_dict()[k].float() for cm in client_models])
        global_dict[k] = stacked.mean(dim=0)
    global_model.load_state_dict(global_dict)


def local_train(model, loader, epochs=2, lr=1e-3, dp=True):
    opt  = torch.optim.Adam(model.parameters(), lr=lr)
    crit = nn.BCELoss()
    model.train()
    for _ in range(epochs):
        for Xb, yb in loader:
            Xb, yb = Xb.to(device), yb.to(device)
            opt.zero_grad()
            loss = crit(model(Xb), yb)
            loss.backward()
            if dp:
                add_dp_noise(model)
            opt.step()
    return model

# CELL 19 — DA-SCT: Domain Adaptation SCT Architecture
# ============================================================
# Combines:
#   • Spiking Convolutional Transformer (SCT) as feature extractor
#   • Gradient Reversal Layer (GRL) for cross-patient domain adaptation
#   • Maximum Mean Discrepancy (MMD) loss for distribution alignment
# ============================================================

class MMDLoss(nn.Module):
    """Maximum Mean Discrepancy loss for domain alignment."""

    def __init__(self, kernel_mul=2.0, kernel_num=5):
        super().__init__()
        self.kernel_mul = kernel_mul
        self.kernel_num = kernel_num

    def gaussian_kernel(self, source, target):
        n_s, n_t = source.size(0), target.size(0)
        total    = torch.cat([source, target], dim=0)
        total_sq = (total.unsqueeze(0) - total.unsqueeze(1)).pow(2).sum(dim=2)
        bandwidth = total_sq.data.sum() / ((n_s + n_t) ** 2 - (n_s + n_t))
        bandwidth /= self.kernel_mul ** (self.kernel_num // 2)
        bw_list    = [bandwidth * (self.kernel_mul ** i) for i in range(self.kernel_num)]
        return sum(torch.exp(-total_sq / bw) for bw in bw_list)

    def forward(self, source, target):
        n_s = source.size(0); n_t = target.size(0)
        kernels = self.gaussian_kernel(source, target)
        XX = kernels[:n_s, :n_s].mean()
        YY = kernels[n_s:, n_s:].mean()
        XY = kernels[:n_s, n_s:].mean()
        return XX + YY - 2 * XY


class DASCT(nn.Module):
    """Domain Adaptation Spiking Convolutional Transformer.
    
    Architecture:
        Input (B,T,C)
        → SpikeEncoder
        → Conv×3 (temporal feature extraction)
        → TransformerEncoder (global temporal context)
        → [Label Classifier]  ← seizure prediction
        → [GRL → Domain Classifier]  ← domain adversarial
    """

    def __init__(self, n_ch=18, T=1024, n_domains=2,
                 d_model=64, n_heads=4, n_layers=2, dropout=0.2):
        super().__init__()

        # ── Spiking feature extractor ───────────────────────
        self.spike_enc = SpikeEncoder()
        self.conv1 = SpikingConvBlock(n_ch,    32, k=7)
        self.conv2 = SpikingConvBlock(32,      64, k=5)
        self.conv3 = SpikingConvBlock(64, d_model, k=3)

        # ── Transformer encoder ─────────────────────────────
        enc_layer = nn.TransformerEncoderLayer(
            d_model=d_model, nhead=n_heads,
            dim_feedforward=d_model * 4, dropout=dropout, batch_first=True
        )
        self.transformer = nn.TransformerEncoder(enc_layer, num_layers=n_layers)
        self.cls_token   = nn.Parameter(torch.zeros(1, 1, d_model))

        # ── Label classifier ────────────────────────────────
        self.label_clf = nn.Sequential(
            nn.Linear(d_model, 32), nn.ReLU(), nn.Dropout(dropout),
            nn.Linear(32, 1), nn.Sigmoid()
        )

        # ── Domain classifier via GRL ────────────────────────
        self.grl = GRL()
        self.domain_clf = nn.Sequential(
            nn.Linear(d_model, 64), nn.ReLU(), nn.Dropout(dropout),
            nn.Linear(64, n_domains)
        )

        # ── MMD loss ─────────────────────────────────────────
        self.mmd = MMDLoss()

    def encode(self, x):
        """Shared encoding: spikes → conv → transformer → CLS embedding."""
        x   = self.spike_enc(x)           # (B,T,C)
        x   = x.permute(0, 2, 1)         # (B,C,T)
        x   = self.conv1(x)
        x   = self.conv2(x)
        x   = self.conv3(x)
        x   = x.permute(0, 2, 1)         # (B,T',d_model)
        cls = self.cls_token.expand(x.size(0), -1, -1)
        x   = torch.cat([cls, x], dim=1)
        x   = self.transformer(x)
        return x[:, 0]                    # CLS embedding (B, d_model)

    def forward(self, x, alpha=1.0, x_target=None):
        feat        = self.encode(x)
        label_out   = self.label_clf(feat).squeeze(-1)

        alpha_t     = torch.tensor(alpha, dtype=torch.float32)
        feat_rev    = self.grl(feat, alpha_t)
        domain_out  = self.domain_clf(feat_rev)

        mmd_loss = torch.tensor(0.0, device=x.device)
        if x_target is not None and x_target.size(0) > 0:
            feat_tgt = self.encode(x_target)
            mmd_loss = self.mmd(feat, feat_tgt)

        return label_out, domain_out, mmd_loss

def train_dasct(model, train_loader, val_loader, epochs=30, lr=1e-3):
    optimizer  = torch.optim.Adam(model.parameters(), lr=lr, weight_decay=1e-4)
    scheduler  = torch.optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=epochs)
    lbl_crit   = nn.BCELoss()
    dom_crit   = nn.CrossEntropyLoss()

    history = {'train_loss': [], 'val_loss': [], 'val_auc': []}
    best_auc, best_state = 0, None

    # Split train_loader to simulate source / target domains
    train_list = list(train_loader)

    print(f"  Training DA-SCT for {epochs} epochs…")
    for ep in range(1, epochs + 1):
        model.train(); ep_loss = 0
        p     = ep / epochs
        alpha = 2.0 / (1 + np.exp(-10 * p)) - 1.0

        for i, (Xb, yb) in enumerate(train_list):
            Xb, yb = Xb.to(device), yb.to(device)
            # Simulate target domain: next batch (circular)
            Xt, _  = train_list[(i + 1) % len(train_list)]
            Xt     = Xt.to(device)

            dom_labels = torch.randint(0, 2, (Xb.size(0),)).to(device)
            optimizer.zero_grad()

            lbl_out, dom_out, mmd_loss = model(Xb, alpha, x_target=Xt)
            lbl_loss = lbl_crit(lbl_out, yb)
            dom_loss = dom_crit(dom_out, dom_labels)
            loss     = lbl_loss + 0.1 * dom_loss + 0.05 * mmd_loss

            loss.backward(); optimizer.step()
            ep_loss += loss.item()
        scheduler.step()

        model.eval(); vloss = 0; vp, vy = [], []
        with torch.no_grad():
            for Xb, yb in val_loader:
                Xb, yb = Xb.to(device), yb.to(device)
                dom_labels = torch.randint(0, 2, (Xb.size(0),)).to(device)
                lbl_out, dom_out, mmd_loss = model(Xb, alpha)
                vloss += (lbl_crit(lbl_out, yb) + 0.1 * dom_crit(dom_out, dom_labels)).item()
                vp.extend(lbl_out.cpu().numpy()); vy.extend(yb.cpu().numpy())
        try:    vauc = roc_auc_score(vy, vp)
        except: vauc = 0.5

        history['train_loss'].append(ep_loss / len(train_list))
        history['val_loss'].append(vloss / len(val_loader))
        history['val_auc'].append(vauc)
        if vauc > best_auc:
            best_auc = vauc; best_state = copy.deepcopy(model.state_dict())
        if ep % 5 == 0:
            print(f"    Ep {ep:03d} | Loss={ep_loss/len(train_list):.4f} | ValAUC={vauc:.4f} | α={alpha:.3f}")

    model.load_state_dict(best_state)
    print(f"  ✅  Best ValAUC = {best_auc:.4f}")
    return history

# ============================================================
# Extends the base GRL model with:
#   • Multi-scale convolutional feature extraction
#   • Attention-based temporal pooling
#   • CORAL (Correlation Alignment) loss for deeper domain alignment
#   • Progressive alpha scheduling per-sample
# ============================================================

class CORALLoss(nn.Module):
    """CORAL: align second-order statistics of source and target."""

    def forward(self, source, target):
        d = source.size(1)
        # Source covariance
        ns = source.size(0)
        cs = (source.t().mm(source) - source.sum(0).unsqueeze(1).mm(
              source.sum(0).unsqueeze(0)) / ns) / (ns - 1 + 1e-8)
        # Target covariance
        nt = target.size(0)
        ct = (target.t().mm(target) - target.sum(0).unsqueeze(1).mm(
              target.sum(0).unsqueeze(0)) / nt) / (nt - 1 + 1e-8)
        loss = (cs - ct).pow(2).sum() / (4 * d * d)
        return loss


class AttentionPool(nn.Module):
    """Soft attention pooling over time dimension."""

    def __init__(self, d_model):
        super().__init__()
        self.attn = nn.Linear(d_model, 1)

    def forward(self, x):
        # x: (B, T, d_model)
        weights = torch.softmax(self.attn(x), dim=1)   # (B, T, 1)
        return (weights * x).sum(dim=1)                 # (B, d_model)


class DAGRLModel(nn.Module):
    """Enhanced Domain Adaptation GRL with multi-scale Conv + Attention + CORAL.
    
    Architecture:
        Input (B,T,C)
        → Multi-scale Conv (3, 5, 7 kernels) → concat → BN → ReLU
        → Attention Temporal Pooling
        → [Label Classifier]
        → [GRL → Domain Classifier]
        CORAL loss between source/target feature distributions
    """

    def __init__(self, n_ch=18, d_model=128, n_domains=2, dropout=0.3):
        super().__init__()

        # ── Multi-scale feature extractor ───────────────────
        self.conv_s = nn.Sequential(
            nn.Conv1d(n_ch, 32, 3, padding=1), nn.BatchNorm1d(32), nn.ReLU(), nn.MaxPool1d(2))
        self.conv_m = nn.Sequential(
            nn.Conv1d(n_ch, 32, 5, padding=2), nn.BatchNorm1d(32), nn.ReLU(), nn.MaxPool1d(2))
        self.conv_l = nn.Sequential(
            nn.Conv1d(n_ch, 32, 7, padding=3), nn.BatchNorm1d(32), nn.ReLU(), nn.MaxPool1d(2))

        fused_ch = 96  # 32*3

        self.deep = nn.Sequential(
            nn.Conv1d(fused_ch, 128, 5, padding=2), nn.BatchNorm1d(128), nn.ReLU(), nn.MaxPool1d(2),
            nn.Conv1d(128, d_model, 3, padding=1), nn.BatchNorm1d(d_model), nn.ReLU(),
        )
        self.attn_pool = AttentionPool(d_model)

        # ── Label classifier ────────────────────────────────
        self.label_clf = nn.Sequential(
            nn.Linear(d_model, 64), nn.ReLU(), nn.Dropout(dropout),
            nn.Linear(64, 16),      nn.ReLU(),
            nn.Linear(16, 1),       nn.Sigmoid()
        )

        # ── Domain classifier via GRL ────────────────────────
        self.grl = GRL()
        self.domain_clf = nn.Sequential(
            nn.Linear(d_model, 64), nn.ReLU(), nn.Dropout(dropout),
            nn.Linear(64, 16),      nn.ReLU(),
            nn.Linear(16, n_domains)
        )

        # ── CORAL loss ───────────────────────────────────────
        self.coral = CORALLoss()

    def encode(self, x):
        """Multi-scale encoding."""
        # x: (B, T, C) → (B, C, T)
        x = x.permute(0, 2, 1)
        f = torch.cat([self.conv_s(x), self.conv_m(x), self.conv_l(x)], dim=1)
        f = self.deep(f)
        f = f.permute(0, 2, 1)          # (B, T', d_model)
        return self.attn_pool(f)        # (B, d_model)

    def forward(self, x, alpha=1.0, x_target=None):
        feat       = self.encode(x)
        label_out  = self.label_clf(feat).squeeze(-1)

        alpha_t    = torch.tensor(alpha, dtype=torch.float32)
        feat_rev   = self.grl(feat, alpha_t)
        domain_out = self.domain_clf(feat_rev)

        coral_loss = torch.tensor(0.0, device=x.device)
        if x_target is not None and x_target.size(0) > 0:
            feat_tgt   = self.encode(x_target)
            coral_loss = self.coral(feat, feat_tgt)

        return label_out, domain_out, coral_loss

    for k, v in metrics.items():
        print(f"    {k:12}: {v:.4f}")
    return metrics, all_probs, all_preds, all_labels

# CELL 26 — XAI STEP 1: Universal SHAP Wrapper
# ============================================================
# Handles all four model output signatures:
#   SCT       → logit scalar  (B,)
#   GRL       → (logit, domain)
#   DA-SCT    → (logit, domain, mmd)
#   DA-GRL    → (logit, domain, coral)
# ============================================================

class SHAPWrapper(nn.Module):
    """SHAP-compatible wrapper for all four model types."""

    def __init__(self, model, model_type='sct'):
        super().__init__()
        self.model = model
        self.model_type = model_type.lower().replace('-', '')

    def forward(self, x):
        mt = self.model_type
        if mt == 'grl':
            out, _ = self.model(x)
        elif mt in ('dasct', 'dagrl'):
            out, _, _ = self.model(x)
        else:  # sct / fed
            out = self.model(x)

        out = torch.as_tensor(out)
        # Ensure (B, 1) shape
        if out.ndim == 1:
            out = out.unsqueeze(1)
        elif out.ndim > 2:
            out = out.reshape(out.shape[0], -1)
        if out.shape[1] > 1:
            out = out[:, :1]
        return out

print("✅  SHAPWrapper defined")

# Extract internal attention weights from the DA-GRL temporal
# attention pooling layer for interpretability.
# ============================================================

def extract_attention_map(model, model_type, X_input, n_samples=10):
    """
    Extract attention weights from model's temporal attention layer.
    Returns: attn_weights (n_samples, T) or None if not available.
    """
    model.cpu().eval()
    attn_hooks = []
    attn_outputs = []

    def hook_fn(module, inp, out):
        # Capture softmax output (attention weights)
        attn_outputs.append(out.detach().cpu().numpy())

    # Register hook on attention layers
    for name, module in model.named_modules():
        if isinstance(module, nn.Softmax):
            attn_hooks.append(module.register_forward_hook(hook_fn))

    X_sample = torch.tensor(X_input[:n_samples], dtype=torch.float32)
    with torch.no_grad():
        mt = model_type.lower().replace('-','')
        if mt == 'grl':
            _ = model(X_sample)
        elif mt in ('dasct', 'dagrl'):
            _ = model(X_sample)
        else:
            _ = model(X_sample)

    for h in attn_hooks:
        h.remove()

    if not attn_outputs:
        print("  ⚠️  No attention layer found — skipping attention plot")
        return None

    # Take last captured attention (temporal pooling)
    attn = attn_outputs[-1]    # shape varies by model
    return attn

from scipy.stats import spearmanr

shap_importance = pd.Series(importance, index=CH_NAMES[:len(importance)])

lime_weights_raw = {feat: wt for feat, wt in exp.as_list()}
lime_ch_agg = np.zeros(len(CH_NAMES))
for feat, wt in lime_weights_raw.items():
    for c_idx, ch in enumerate(CH_NAMES):
        if ch in feat:
            lime_ch_agg[c_idx] += abs(wt)

lime_importance = pd.Series(lime_ch_agg, index=CH_NAMES)
lime_importance = lime_importance / (lime_importance.max() + 1e-9)

common_channels = shap_importance.index.intersection(lime_importance.index)
shap_vec = shap_importance[common_channels].values
lime_vec = lime_importance[common_channels].values

rho, pval = spearmanr(shap_vec, lime_vec)
print("="*60); print("⚖️   SHAP vs LIME COMPARATIVE ANALYSIS"); print("="*60)
print(f"  Spearman ρ = {rho:.4f}  (p={pval:.4f})")
print(f"  Agreement  : {'High' if abs(rho)>0.7 else 'Moderate' if abs(rho)>0.4 else 'Low'}")

top5_shap = shap_importance.nlargest(5).index.tolist()
top5_lime = lime_importance.nlargest(5).index.tolist()
overlap   = set(top5_shap) & set(top5_lime)
jaccard   = len(overlap) / len(set(top5_shap) | set(top5_lime))

print(f"\n  Top-5 SHAP channels : {top5_shap}")
print(f"  Top-5 LIME channels : {top5_lime}")
print(f"  Overlap             : {sorted(overlap)}")
print(f"  Jaccard similarity  : {jaccard:.4f}")
